declare -A seen
order=()
while read -r w || [ -n "$w" ]; do
  [ -n "$w" ] || continue
  c=${seen[$w]:-0}
  if [ "$c" -eq 0 ]; then
    order+=("$w")
  fi
  seen[$w]=$((c + 1))
done
for w in "${order[@]}"; do
  echo "$w ${seen[$w]}"
done
