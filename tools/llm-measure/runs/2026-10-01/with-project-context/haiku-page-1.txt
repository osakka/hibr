declare -A seen
declare -a order
while read word; do
  if [ -z "${seen[$word]}" ]; then
    seen[$word]=1
    order+=("$word")
  else
    seen[$word]=$((seen[$word] + 1))
  fi
done
for w in "${order[@]}"; do
  echo "$w ${seen[$w]}"
done
