order=()
while read -r w || [ -n "$w" ]; do
  [ -z "$w" ] && continue
  if [ -z "${seen[$w]}" ]; then
    order+=("$w")
    seen[$w]=1
  else
    c=${seen[$w]}
    seen[$w]=$((c + 1))
  fi
done
for w in "${order[@]}"; do
  echo "$w ${seen[$w]}"
