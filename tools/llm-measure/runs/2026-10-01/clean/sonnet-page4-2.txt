#!/usr/bin/env hibr
declare -A count
order=()
while read -r w; do
  if [ -z "${count[$w]}" ]; then
    order+=("$w")
  fi
  count[$w]=$(( ${count[$w]:-0} + 1 ))
done
for w in "${order[@]}"; do
  echo "$w ${count[$w]}"
done
