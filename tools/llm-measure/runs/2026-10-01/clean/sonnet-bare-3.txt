=== T3
declare -A count
order=()
while IFS= read -r word; do
    [[ -z $word ]] && continue
    if [[ -z ${count[$word]+x} ]]; then
        order+=("$word")
        count[$word]=0
    fi
    count[$word]=$(( count[$word] + 1 ))
done
for w in "${order[@]}"; do
    echo "$w ${count[$w]}"
done

n=${#args[@]}
for (( i = 1; i < n; i++ )); do
    key=${args[i]}
    j=$(( i - 1 ))
    while (( j >= 0 && args[j] > key )); do
        args[j+1]=${args[j]}
        j=$(( j - 1 ))
    done
    args[j+1]=$key
done
