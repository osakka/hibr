declare -A seen
order=()
while read -r w; do
  if [ -z "${seen[$w]}" ]; then
    seen[$w]=1
    arr push order "$w"
  else
    seen[$w]=$((seen[$w] + 1))
  fi
done
for w in "${order[@]}"; do
  echo "$w ${seen[$w]}"
done
