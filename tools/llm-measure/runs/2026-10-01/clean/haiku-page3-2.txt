declare -A seen
declare -a order
while read -r word; do
  if [[ -z "${seen[$word]}" ]]; then
    seen[$word]=1
    order+=("$word")
  else
    seen[$word]=$((seen[$word] + 1))
  fi
done
for word in "${order[@]}"; do
  echo "$word ${seen[$word]}"
done
