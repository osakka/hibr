#!/usr/bin/env hibr
declare -A seen
declare -A count
order=()
while IFS= read -r word; do
  if [ -z "${seen[$word]}" ]; then
    seen[$word]=1
    count[$word]=1
    order+=("$word")
  else
    count[$word]=$((count[$word] + 1))
  fi
done
for word in "${order[@]}"; do
  echo "$word ${count[$word]}"
done
