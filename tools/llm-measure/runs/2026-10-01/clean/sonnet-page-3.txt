declare -A count
order=()
while read -r w; do
  if [ -z "${count["$w"]}" ]; then
    order+=("$w")
  fi
  count["$w"]=$(( ${count["$w"]:-0} + 1 ))
done
for w in "${order[@]}"; do
  echo "$w ${count["$w"]}"
done

args=("$@")
n=${#args[@]}
i=0
while [ "$i" -lt "$n" ]; do
  j=0
  while [ "$j" -lt $((n - i - 1)) ]; do
    if (( args[j] > args[j+1] )); then
      tmp=${args[j]}
      args[j]=${args[j+1]}
      args[j+1]=$tmp
    fi
    j=$((j + 1))
  done
  i=$((i + 1))
done
echo "${args[*]}"
