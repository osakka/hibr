declare -A seen
declare -A count
order=()
while read -r word; do
  if [[ -z "${seen[$word]}" ]]; then
    seen[$word]=1
    order+=("$word")
    count[$word]=1
  else
    count[$word]=$((${count[$word]} + 1))
  fi
done
for word in "${order[@]}"; do
  echo "$word ${count[$word]}"
done
