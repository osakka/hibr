declare -a order
while read word; do
  if [[ ! -v counts[$word] ]]; then
    order+=("$word")
    counts[$word]=0
  fi
  ((counts[$word]++))
done
for word in "${order[@]}"; do
  echo "$word ${counts[$word]}"
done
