declare -A seen
order=()
while read -r word; do
  if [ -z "${seen[$word]}" ]; then
    seen[$word]=1
    order+=("$word")
  else
    seen[$word]=$((${seen[$word]} + 1))
  fi
done
for w in "${order[@]}"; do
  echo "$w ${seen[$w]}"
done
