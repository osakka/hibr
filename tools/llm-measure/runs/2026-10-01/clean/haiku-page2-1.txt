declare -A counts
order=()
while read -r word; do
  if [ -z "${counts[$word]}" ]; then
    order+=("$word")
    counts[$word]=1
  else
    counts[$word]=$((counts[$word] + 1))
  fi
done
for word in "${order[@]}"; do
  echo "$word ${counts[$word]}"
done

  opt .  "Program"
  opt -v --verbose verbose "Verbose flag"
  opt -n --count count int=1 "Count"
  args "${argv[@]}" || return
  echo "verbose=$verbose count=$count rest=${ARGS[*]}"
}
