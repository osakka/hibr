declare -A counts
order=()
while read -r w; do
  if [ -z "${counts[$w]}" ]; then
    order+=("$w")
    counts[$w]=1
  else
    counts[$w]=$((${counts[$w]} + 1))
  fi
done
for w in "${order[@]}"; do
  echo "$w ${counts[$w]}"
done

  opt .  "Process arguments"
  opt -v --verbose verbose "Verbose mode"
  opt -n --count count int=1 "Count"
  args "${argv[@]}" || return
  echo "verbose=$verbose count=$count rest=${ARGS[*]}"
}
