=== T3
declare -A seen
order=()
while IFS= read -r word; do
    if [[ ! -v seen[$word] ]]; then
        seen[$word]=0
        order+=("$word")
    fi
    ((seen[$word]++))
done
for word in "${order[@]}"; do
    echo "$word ${seen[$word]}"
done
