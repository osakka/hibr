declare -A seen
declare -A count
order=()
while read -r w; do
  if [ -z "${seen[$w]}" ]; then
    seen[$w]=1
    order+=("$w")
    count[$w]=1
  else
    count[$w]=$((count[$w] + 1))
  fi
done
for w in "${order[@]}"; do
  echo "$w ${count[$w]}"
done
