=== T2
declare -A m=( [a]=1 [b]=2 [c]=3 )
k=b
unset "m[$k]"
keys=("${!m[@]}")
n=${#keys[@]}
for (( i = 0; i < n; i++ )); do
    for (( j = i + 1; j < n; j++ )); do
        if [[ "${keys[j]}" < "${keys[i]}" ]]; then
            tmp=${keys[i]}
            keys[i]=${keys[j]}
            keys[j]=$tmp
        fi
    done
done
out=""
for key in "${keys[@]}"; do
    if [[ -z $out ]]; then
        out=$key
    else
        out="$out $key"
    fi
done
echo "$out"

=== T3
declare -A count
order=()
while IFS= read -r word || [[ -n $word ]]; do
    [[ -z $word ]] && continue
    if [[ -z ${count[$word]+x} ]]; then
        order+=("$word")
        count[$word]=1
    else
        count[$word]=$(( ${count[$word]} + 1 ))
    fi
done
for word in "${order[@]}"; do
    echo "$word ${count[$word]}"
done
